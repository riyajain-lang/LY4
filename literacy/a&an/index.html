.sentence-box img { width: 200px; /* increased from 150px */ height: 200px; object-fit:contain; } /* Confetti JS update */ function confetti(){ const canvas = document.getElementById('confettiCanvas'); const ctx = canvas.getContext('2d'); canvas.width = window.innerWidth; canvas.height = window.innerHeight; const confettis=[]; for(let i=0;i<200;i++){ /* increased number of confetti particles */ confettis.push({ x:Math.random()*canvas.width, y:Math.random()*canvas.height-canvas.height, r:Math.random()*10+6, /* increased size range */ d:Math.random()*150+50, color:Math.random()<0.5?"#FFB6C1":"#FFFFE0", tilt:Math.random()*15-10 /* increased tilt for more spread */ }); } function draw(){ ctx.clearRect(0,0,canvas.width,canvas.height); confettis.forEach(c=>{ ctx.beginPath(); ctx.lineWidth=c.r/2; ctx.strokeStyle=c.color; ctx.moveTo(c.x+c.tilt+c.r/4,c.y); ctx.lineTo(c.x+c.tilt,c.y+c.tilt+c.r/4); ctx.stroke(); c.y+=3; if(c.y>canvas.height) c.y=-10; /* faster falling */ }); requestAnimationFrame(draw); } draw(); }
